    function getFilteredDishes() {
      return MENU_ITEMS.filter(dish => {
        // Category filter
        if (state.selectedCategory !== 'All' && dish.category !== state.selectedCategory) {
          return false;
        }
        // Dietary filter
        if (state.dietaryFilter === 'veg' && !dish.veg) return false;
        if (state.dietaryFilter === 'non-veg' && dish.veg) return false;
        if (state.dietaryFilter === 'special' && !dish.special) return false;

        // Search query
        if (state.searchQuery) {
          const haystack = `${dish.name} ${dish.description} ${dish.category}`.toLowerCase();
          if (!haystack.includes(state.searchQuery)) return false;
        }
        return true;
      });
    }

    function updateQuantity(dishId, delta) {
      const currentQty = state.cart[dishId] || 0;
      const nextQty = currentQty + delta;
      const dish = MENU_ITEMS.find(d => d.id === dishId);

      if (nextQty <= 0) {
        delete state.cart[dishId];
        if (dish) showToast(`Removed ${dish.name}`);
      } else {
        state.cart[dishId] = nextQty;
        if (currentQty === 0 && dish) {
          showToast(`Added ${dish.name} to order`);
        }
      }

      renderMenu();
      renderCartUI();
    }

    function updateInstructions(val) {
      state.specialInstructions = val;
      // Sync between desktop sidebar textarea and mobile drawer textarea if both exist
      document.querySelectorAll('.special-instructions-input').forEach(el => {
        if (el.value !== val) el.value = val;
      });
    }

    function getCartSummary() {
      let totalItems = 0;
      let subtotal = 0;
      const items = [];

      for (const [dishId, qty] of Object.entries(state.cart)) {
        const dish = MENU_ITEMS.find(d => d.id === dishId);
        if (dish && qty > 0) {
          const lineTotal = dish.price * qty;
          totalItems += qty;
          subtotal += lineTotal;
          items.push({ ...dish, qty, lineTotal });
        }
      }

      const gst = Math.round(subtotal * 0.05);
      const grandTotal = subtotal + gst;
      return { items, totalItems, subtotal, gst, grandTotal };
    }

    // Input formatting helpers for Card inputs
    function formatCardNumber(input) {
      let v = input.value.replace(/\D/g, '').substring(0, 16);
      input.value = v.replace(/(.{4})/g, '$1 ').trim();
    }

    function formatCardExpiry(input) {
      let v = input.value.replace(/\D/g, '').substring(0, 4);
      if (v.length >= 3) {
        input.value = `${v.substring(0, 2)}/${v.substring(2, 4)}`;
      } else {
        input.value = v;
      }
    }

    function processOrderPayment() {
      const summary = getCartSummary();
      if (summary.totalItems === 0) return;

      // Validate active payment method inputs
      if (state.paymentMethod === 'upi') {
        const upiVal = document.getElementById('upiIdInput').value.trim();
        const upiErr = document.getElementById('upiErrorMsg');
        if (!state.selectedUpiApp && (!upiVal || !upiVal.includes('@'))) {
          upiErr.classList.remove('hidden');
          return;
        }
        upiErr.classList.add('hidden');
      } else if (state.paymentMethod === 'card') {
        const cardNum = document.getElementById('cardNumberInput').value.replace(/\s/g, '');
        const cardExp = document.getElementById('cardExpiryInput').value.trim();
        const cardCvv = document.getElementById('cardCvvInput').value.trim();
        const cardErr = document.getElementById('cardErrorMsg');
        if (cardNum.length < 12 || cardExp.length < 4 || cardCvv.length < 3) {
          cardErr.classList.remove('hidden');
          return;
        }
        cardErr.classList.add('hidden');
      }

      // Simulate brief payment processing state
      const btn = document.getElementById('confirmPaymentBtn');
      const btnText = document.getElementById('confirmPaymentBtnText');
      btn.disabled = true;
      btnText.textContent = 'Processing Order...';

      setTimeout(() => {
        btn.disabled = false;

        // Generate Order ID & payment label
        const randomNum = Math.floor(1000 + Math.random() * 9000);
        const orderId = `SE-${randomNum}`;
        let paymentLabel = 'Paid via UPI';
        if (state.paymentMethod === 'upi' && state.selectedUpiApp) {
          paymentLabel = `Paid via ${state.selectedUpiApp}`;
        } else if (state.paymentMethod === 'card') {
          paymentLabel = 'Paid via Card';
        } else if (state.paymentMethod === 'cash') {
          paymentLabel = 'Pay at Table';
        }

        const now = new Date();
        const timeStr = now.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

        // Save Active Order
        state.activeOrder = {
          orderId,
          tableNumber: state.tableNumber,
          items: summary.items,
          subtotal: summary.subtotal,
          gst: summary.gst,
          grandTotal: summary.grandTotal,
          instructions: state.specialInstructions,
          paymentLabel,
          timestamp: timeStr,
          stage: 2 // 1 = Placed, 2 = Preparing, 3 = Ready to Serve
        };

        // Clear current cart after successful placement
        state.cart = {};
        state.specialInstructions = '';
        renderMenu();
        renderCartUI();

        // Show "Live Order" button in top navbar
        const navTrackerBtn = document.getElementById('activeOrderNavBtn');
        navTrackerBtn.classList.remove('hidden');
        navTrackerBtn.classList.add('flex');

        closeCheckoutModal();
        openTrackerModal();
        startAutoStageProgression();
      }, 650);
    }

    function openTrackerModal() {
      if (!state.activeOrder) return;
      const order = state.activeOrder;

      document.getElementById('trackerTableLabel').textContent = `Table #${order.tableNumber}`;
      document.getElementById('trackerOrderId').textContent = `Order #${order.orderId}`;
      document.getElementById('trackerPaymentBadge').textContent = order.paymentLabel;
      document.getElementById('receiptTimestamp').textContent = order.timestamp;

      // Render itemized digital receipt
      document.getElementById('receiptItemsList').innerHTML = order.items.map(item => `
        <div class="flex items-center justify-between text-xs sm:text-sm">
          <span class="text-stone-800"><strong class="font-semibold">${item.qty}×</strong> ${item.name}</span>
          <span class="font-semibold text-stone-900">₹${item.lineTotal}</span>
        </div>
      `).join('');

      const noteBox = document.getElementById('receiptSpecialNote');
      if (order.instructions) {
        noteBox.textContent = `Note: "${order.instructions}"`;
        noteBox.classList.remove('hidden');
      } else {
        noteBox.classList.add('hidden');
      }

      document.getElementById('receiptSubtotal').textContent = `₹${order.subtotal}`;
      document.getElementById('receiptGst').textContent = `₹${order.gst}`;
      document.getElementById('receiptTotal').textContent = `₹${order.grandTotal}`;

      renderOrderStageUI();

      const modal = document.getElementById('trackerModal');
      modal.classList.remove('hidden');
      modal.classList.add('flex');
    }

    function closeTrackerModal() {
      const modal = document.getElementById('trackerModal');
      modal.classList.add('hidden');
      modal.classList.remove('flex');
    }

    function renderOrderStageUI() {
      if (!state.activeOrder) return;
      const stage = state.activeOrder.stage;

      const step2Circle = document.getElementById('step2Circle');
      const step3 = document.getElementById('step-3');
      const step3Circle = document.getElementById('step3Circle');
      const estTime = document.getElementById('trackerEstTime');
      const simBtn = document.getElementById('simulateStageBtn');

      if (stage === 2) {
        step2Circle.className = 'w-8 h-8 rounded-full bg-amber-500 text-white flex items-center justify-center text-xs font-bold shrink-0 animate-soft-pulse';
        step2Circle.textContent = '2';
        step3.classList.add('opacity-45');
        step3Circle.className = 'w-8 h-8 rounded-full bg-stone-300 text-stone-700 flex items-center justify-center text-xs font-bold shrink-0';
        step3Circle.textContent = '3';
        estTime.textContent = '15–20 Mins';
        simBtn.classList.remove('hidden');
      } else if (stage >= 3) {
        step2Circle.className = 'w-8 h-8 rounded-full bg-emerald-600 text-white flex items-center justify-center text-xs font-bold shrink-0';
        step2Circle.textContent = '✓';
        step3.classList.remove('opacity-45');
        step3Circle.className = 'w-8 h-8 rounded-full bg-emerald-600 text-white flex items-center justify-center text-xs font-bold shrink-0 animate-soft-pulse';
        step3Circle.textContent = '✓';
        estTime.textContent = 'Serving Now at Table!';
        simBtn.classList.add('hidden');
      }
    }

    function advanceOrderStage() {
      if (!state.activeOrder) return;
      if (state.activeOrder.stage < 3) {
        state.activeOrder.stage = 3;
        renderOrderStageUI();
        showToast(`Order #${state.activeOrder.orderId} is Ready to Serve!`);
      }
    }

    function startAutoStageProgression() {
      if (state.orderStageTimer) clearTimeout(state.orderStageTimer);
      // Automatically transition from Preparing -> Ready to Serve after 18 seconds for live realism
      state.orderStageTimer = setTimeout(() => {
        if (state.activeOrder && state.activeOrder.stage < 3) {
          advanceOrderStage();
        }
      }, 18000);
    }

    // -----------------------------------------------------------------------
    // 9. TOAST NOTIFICATION HELPER
    // -----------------------------------------------------------------------
    let toastTimeout = null;
    function showToast(message) {
      const toast = document.getElementById('toastNotification');
      document.getElementById('toastMessage').textContent = message;
      toast.classList.remove('opacity-0', '-translate-y-4');
      if (toastTimeout) clearTimeout(toastTimeout);
      toastTimeout = setTimeout(() => {
        toast.classList.add('opacity-0', '-translate-y-4');
      }, 2400);
    }
